  ```python
      """Resolve {endpoint, token} from toml > env > default."""
      raw: dict = {}
      path = _find_config()

                 or _BLOBSTORE_DEFAULT_ENDPOINT
      token = raw.get("token") or os.environ.get("IMAGECLI_BLOBSTORE_TOKEN")
      return {"endpoint": endpoint, "token": token}
  ```
